数据库反应器

数据库反应器是用于接收与数据库状态相关的事件，例如对象被加入到数据库，或数据库中的对象被删除、修改等， 这些事件的发送者都是数据库，故我们定义的这类事件都会被加入到数据库事件列表中。

常用数据库反应器

反应器 功能概述
ObjectAppended 添加实体
ObjectErased 删除实体
ObjectModified 修改实体
ObjectOpenedForModify 打开以修改实体

定义反应器

在工程中新建类，设置基类为ZcDbDatabaseReactor，在类中添加所需的事件虚函数。

// DBReactor.h
class DBReactor : public ZcDbDatabaseReactor
{
public:
    ZCRX_DECLARE_MEMBERS(DBReactor);
    DBReactor();
    ~DBReactor();

    //when the object has been added to the database
    virtual void objectAppended(const ZcDbDatabase* dwg, const ZcDbObject* dbObj);
    //when opend by writing
    virtual void objectOpenedForModify(const ZcDbDatabase* dwg, const ZcDbObject* dbObj);
    //when the object is modified
    virtual void objectModified(const ZcDbDatabase* dwg, const ZcDbObject* dbObj);
    //when the object is erased or unerased
    virtual void objectErased(const ZcDbDatabase* dwg, const ZcDbObject* dbObj, ZSoft::Boolean bErased);
};

// DBReactor.cpp
ZCRX_NO_CONS_DEFINE_MEMBERS(DBReactor, ZcDbDatabaseReactor);

DBReactor::DBReactor()
{
}

DBReactor::~DBReactor()
{
}

void DBReactor::objectAppended(const ZcDbDatabase * dwg, const ZcDbObject * dbObj)
{
    zcutPrintf(_T("\nobjectAppended"));
}

void DBReactor::objectOpenedForModify(const ZcDbDatabase * dwg, const ZcDbObject * dbObj)
{
    zcutPrintf(_T("\nobjectOpenedForModify"));
}

void DBReactor::objectModified(const ZcDbDatabase * dwg, const ZcDbObject * dbObj)
{
    zcutPrintf(_T("\nobjectModified"));
}

void DBReactor::objectErased(const ZcDbDatabase * dwg, const ZcDbObject * dbObj, ZSoft::Boolean bErased)
{

添加和删除数据库事件

// 设置全局变量
static DBReactor* g_dbReactor = NULL;
void initApp()
{ 
    // 在zrx初始化入口中注册反应器类
    DBReactor::rxInit();
    zcrxBuildClassHierarchy();
    zcedRegCmds->addCommand(_T("DATABASEREACTOR_COMMANDS"), _T("AddDBReactor"), _T("AddDBReactor"), ZCRX_CMD_MODAL, AddDBReactor);
    zcedRegCmds->addCommand(_T("DATABASEREACTOR_COMMANDS"), _T("RemoveReactor"), _T("RemoveReactor"), ZCRX_CMD_MODAL, RemoveReactor);
}

void unloadApp()
{ 
    RemoveReactor();
    // 删除反应器类
    deleteZcRxClass(DBReactor::desc());
    zcedRegCmds->removeGroup(_T("DATABASEREACTOR_COMMANDS"));
}

void AddDBReactor()
{
    if (g_dbReactor == NULL)
    {
        g_dbReactor = new DBReactor();
        zcdbHostApplicationServices()->workingDatabase()->addReactor(g_dbReactor);
    }
    else
    {
        zcdbHostApplicationServices()->workingDatabase()->addReactor(g_dbReactor);
    }
}

void RemoveReactor()
{
    if (g_dbReactor != NULL)
    {
        zcdbHostApplicationServices()->workingDatabase()->removeReactor(g_dbReactor);
    }
}
