数据库对象的复制

本章节将介绍如何复制数据库对象。数据库对象的复制分为DeepClone和WblockClone两种，分别对应同一个数据库内的复制和不同数据库之间的复制操作。

下表列举了进行数据库对象的复制操作需要用到的接口：

接口 作用
ZcDbDatabase::deepCloneObjects 在同一个数据库中复制对象
ZcDbDatabase::wblock 把对象复制到新的数据库
ZcDbDatabase::wblockCloneObjects 把对象复制到另一个数据库

代码示例

1. 在同一个数据库中复制对象

void DeepCloneDiffOwnerObjects()
{
	zds_name sssetMS, sssetPS;
	ZdInt32 lengthMS = 0, lengthPS = 0;
	const ZTCHAR* promptPtrs1[2] = { _T("\nSelect entities to clone(clone to the model space): ") , _T("") };
	const ZTCHAR* promptPtrs2[2] = { _T("\nSelect entities to clone(clone to the paper space): "), _T("") };

	if (RTNORM != zcedSSGet(_T(":$"), promptPtrs1, NULL, NULL, sssetMS))
		return;
	zcedSSLength(sssetMS, &lengthMS);

	if (RTNORM != zcedSSGet(_T(":$"), promptPtrs2, NULL, NULL, sssetPS))
		return;
	zcedSSLength(sssetPS, &lengthPS);

	ZcDbObjectId ownerId = ZcDbObjectId::kNull;
	ZcDbObjectIdArray idListMS;
	for (int i = 0; i < lengthMS; i++)
	{
		zds_name ent;
		zcedSSName(sssetMS, i, ent);
		ZcDbObjectId objId;
		zcdbGetObjectId(objId, ent);

		ZcDbObject* pObj = NULL;
		zcdbOpenObject(pObj, objId, ZcDb::kForRead);
		if (pObj->ownerId() == ownerId)
		{
			idListMS.append(objId);
		}
		else if (i == 0)
		{
			ownerId = pObj->ownerId();
			idListMS.append(objId);
		}
		pObj->close();
	}
	zcedSSFree(sssetMS);

	ownerId = ZcDbObjectId::kNull;
	ZcDbObjectIdArray idListPS;
	for (int i = 0; i < lengthPS; i++)
	{
		zds_name ent;
		zcedSSName(sssetPS, i, ent);
		ZcDbObjectId objId;
		zcdbGetObjectId(objId, ent);

		ZcDbObject* pObj = NULL;
		zcdbOpenObject(pObj, objId, ZcDb::kForRead);
		if (pObj->ownerId() == ownerId)
		{
			idListPS.append(objId);
		}
		else if (i == 0)
		{
			ownerId = pObj->ownerId();
			idListPS.append(objId);
		}
		pObj->close();
	}
	zcedSSFree(sssetPS);

	ZcDbBlockTable* pBlockTable = NULL;
	zcdbHostApplicationServices()->workingDatabase()->getSymbolTable(pBlockTable, ZcDb::kForRead);
	ZcDbObjectId modelSpaceId, paperSpaceId;
	pBlockTable->getAt(ZCDB_MODEL_SPACE, modelSpaceId);
	pBlockTable->getAt(ZCDB_PAPER_SPACE, paperSpaceId);
	pBlockTable->close();

	ZcDbIdMapping idMap;
	zcdbHostApplicationServices()->workingDatabase()->deepCloneObjects(idListMS, modelSpaceId, idMap);
	zcdbHostApplicationServices()->workingDatabase()->deepCloneObjects(idListPS, paperSpaceId, idMap);

	ZcDbIdMappingIter iter(idMap);
	for (iter.start(); !iter.done(); iter.next())
	{
		ZcDbIdPair idPair;
		iter.getMap(idPair);
		if (!idPair.isCloned())
		{
			continue;
		}

		zcutPrintf(_T("\nID of object：%Ld"), idPair.value().asOldId());
	}
}

注意：需要添加头文件：#include"zdbidmap.h"

运行结果：

让用户分别选择要复制到同一个数据库的模型空间和图纸空间的实体，然后进行复制，最后命令行输出新实体的id。

2. 复制到新的数据库中

 void wblockSample()
{
	// step1: choose dwg to wblock
	resbuf* rb = zcutNewRb(RTSTR);
	if (RTNORM != zcedGetFileD(_T("Choose the dwg file"), NULL, _T("dwg"), 0, rb))
	{
		return;
	}
	ZTCHAR result[MAX_PATH];
	if (RTNORM == zcedFindFile(rb->resval.rstring, result))
	{
		ZcDbDatabase* pDb = new ZcDbDatabase(false, true); // should use null database, or wblock returns ePermanentlyErased
		Zcad::ErrorStatus es = pDb->readDwgFile(result);

		// step2: Get the entity set in model space
		ZcDbObjectIdArray ids;
		getMsEnts(pDb, ids);

		// step3: wblock to the target db
		ZcDbDatabase* pDestDb = NULL;
		es = pDb->wblock(pDestDb, ids, ZcGePoint3d::kOrigin);
		if (es == Zcad::eOk)
		{
			zcutPrintf(_T("\n%d entities are imported."), ids.length());
		}
		else
		{
			delete pDb;
			delete pDestDb;
			zcutRelRb(rb);
			zcutPrintf(_T("\nError: %s"), zcadErrorStatusText(es));
			return;
		}

		// step4: save the target db as dwg file
		CString strToDwgFilePath;
		TCHAR szDocPath[MAX_PATH] = { 0 };
		SHGetFolderPath(NULL, CSIDL_PERSONAL, NULL, SHGFP_TYPE_CURRENT, szDocPath);
		strToDwgFilePath.Format(_T("%s\\TestWblock.dwg"), szDocPath);  // C:\Users\...\Documents\TestWblock.dwg

		es = pDestDb->saveAs(strToDwgFilePath);
		if (es != Zcad::eOk)
		{
			zcutPrintf(_T("\nSave faild, error: %s"), zcadErrorStatusText(es));
		}
		else
		{
			zcutPrintf(_T("\nThe new dwg path: %s"), strToDwgFilePath.GetString());
		}

		delete pDb;
		delete pDestDb;
	}
	zcutRelRb(rb);
}

运行结果：

让用户选择一个DWG文件，把该文件中模型空间的所有实体复制到新的DWG文件中

3. 在两个数据库之间进行复制

   void wblockCloneObjectsSample()
{
	ZcDbBlockTable* pTbl = NULL;
	zcdbCurDwg()->getBlockTable(pTbl, ZcDb::kForRead);
	if (pTbl)
	{
		ZcDbObjectId msId;
		pTbl->getAt(ZCDB_MODEL_SPACE, msId);
		pTbl->close();

		resbuf* rb = zcutNewRb(RTSTR);
		if (RTNORM != zcedGetFileD(_T("Choose the dwg file"), NULL, _T("dwg"), 0, rb))
		{
			return;
		}
		ZTCHAR result[MAX_PATH];
		if (RTNORM == zcedFindFile(rb->resval.rstring, result))
		{
			ZcDbDatabase* pDb = new ZcDbDatabase();
			pDb->readDwgFile(result);

			// Get entity set in model space
			ZcDbObjectIdArray ids;
			getMsEnts(pDb, ids);

			// wblockCloneObjects to current db
			ZcDbIdMapping idMap;
			Zcad::ErrorStatus es = zcdbCurDwg()->wblockCloneObjects(ids, msId, idMap, ZcDb::kDrcReplace);
			if (es == Zcad::eOk)
			{
				zcutPrintf(_T("\n%d entities are imported."), ids.length());
			}
			else
			{
				zcutPrintf(_T("\nError: %s"), zcadErrorStatusText(es));
			}

			delete pDb;
			pDb = NULL;
		}
		zcutRelRb(rb);
	}

}

运行结果：

让用户选择一个DWG文件，把该文件中模型空间的所有实体复制到当前的活动文档中。
