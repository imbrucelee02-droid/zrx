文档反应器

文档反应器用于接收CAD软件层级的特殊事件，例如图纸创建、销毁、被激活等。

常用文档事件

事件 功能概述
documentCreated 创建新文档
documentDestroyed 文档被关闭
documentActivated 文档被激活

定义反应器内容

在工程中新建类，设置基类为ZcApDocManagerReactor，在类中添加所需的事件虚函数。 

// DocManReactor.h
class DocManReactor : public ZcApDocManagerReactor
{
public:
    ZCRX_DECLARE_MEMBERS(DocManReactor);

    DocManReactor();
    ~DocManReactor();

    virtual void documentCreated(ZcApDocument* pDocCreating);
    virtual void documentDestroyed(const ZTCHAR* fileName);
    virtual void documentActivated(ZcApDocument* pActivatedDoc);
};

// DocManReactor.cpp
ZCRX_NO_CONS_DEFINE_MEMBERS(DocManReactor, ZcApDocManagerReactor);

DocManReactor::DocManReactor()
{
}

DocManReactor::~DocManReactor()
{
}

void DocManReactor::documentCreated(ZcApDocument * pDocCreating)
{
	zcutPrintf(_T("\ndocumentCreated"));
}

void DocManReactor::documentDestroyed(const ZTCHAR * fileName)
{
	zcutPrintf(_T("\ndocumentDestroyed"));
}

void DocManReactor::documentActivated(ZcApDocument * pActivatedDoc)
{
	zcutPrintf(_T("\ndocumentActivated"));
}

添加和删除文档反应器

static DocManReactor* g_DocManReactor = NULL;

void initApp()
{ 
    // 注册反应器类
    DocManReactor::rxInit();
    zcrxBuildClassHierarchy();
    zcedRegCmds->addCommand(_T("DOCMANAGERREACTOR_COMMANDS"), _T("AddDocManReactor"), _T("AddDocManReactor"), ZCRX_CMD_MODAL, AddDocManReactor);
    zcedRegCmds->addCommand(_T("DOCMANAGERREACTOR_COMMANDS"), _T("RemoveDocManReactor"), _T("RemoveDocManReactor"), ZCRX_CMD_MODAL, RemoveDocManReactor);
}


void unloadApp()
{ 
    RemoveDocManReactor();
    // 删除反应器类
    deleteZcRxClass(DocManReactor::desc());
    zcedRegCmds->removeGroup(_T("DOCMANAGERREACTOR_COMMANDS"));
}


void AddDocManReactor()
{
    if (g_DocManReactor == NULL)
    {
        g_DocManReactor = new DocManReactor();
        zcDocManager->addReactor(g_DocManReactor);
    }
    else
    {
        zcDocManager->addReactor(g_DocManReactor);
    }
}

void RemoveDocManReactor()
{
    if (g_DocManReactor != NULL)
    {
        zcDocManager->removeReactor(g_DocManReactor);
    }
}
