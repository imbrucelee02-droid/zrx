常用实体的创建和编辑方法

本章节将介绍一些常用的实体，如直线、圆（弧）、椭圆（弧）、多段线、文字、填充、表格、标注等等的创建方法。

直线（Line）

下表列举了用于创建和编辑直线的常用接口：

接口 作用
ZcDbLine( const ZcGePoint3d& start, const ZcGePoint3d& end ); 使用起点和终点构造一条直线
Zcad::ErrorStatus setStartPoint(const ZcGePoint3d&); 设置直线的终点
ZcGePoint3d startPoint() const; 获取起点
Zcad::ErrorStatus setEndPoint(const ZcGePoint3d&); 设置直线的终点
ZcGePoint3d endPoint() const; 获取终点

代码示例

1. 创建直线

void drawLine()
{ 
	ZcGePoint3d startPt(4.0, 2.0, 0.0);
	ZcGePoint3d endPt(10.0, 7.0, 0.0);
	ZcDbLine *pLine = new ZcDbLine(startPt, endPt);

	ZcDbBlockTable *pBlockTable;
	zcdbHostApplicationServices()->workingDatabase()->getSymbolTable(pBlockTable, ZcDb::kForRead);

	ZcDbBlockTableRecord *pBlockTableRecord;
	pBlockTable->getAt(ZCDB_MODEL_SPACE, pBlockTableRecord,	ZcDb::kForWrite);
	pBlockTable->close();

	ZcDbObjectId lineId;
	pBlockTableRecord->appendZcDbEntity(lineId, pLine);

	pBlockTableRecord->close();
	pLine->close();
}

运行结果：

在模型空间创建一条直线。

2. 编辑直线

void ModLine()
{
	ZcDbDatabase* pDb = zcdbHostApplicationServices()->workingDatabase();
	if (pDb == NULL)
	{
		return;
	}

	ZcDbBlockTable* pBlockTable = NULL;
	if (pDb->getSymbolTable(pBlockTable, ZcDb::kForRead) != Zcad::eOk)
	{
		return;
	}

	ZcDbBlockTableRecord* pBlockRec = NULL;
	if (pBlockTable->getAt(ZCDB_MODEL_SPACE, pBlockRec, ZcDb::kForRead) != Zcad::eOk)
	{
		pBlockTable->close();
		return;
	}

	ZcDbBlockTableRecordIterator * pIterator = NULL;
	pBlockRec->newIterator(pIterator);
	pBlockRec->close();
	pBlockTable->close();

	for (; !pIterator->done(); pIterator->step())
	{
		ZcDbEntity * pEntity = NULL;
		Zcad::ErrorStatus esGet = pIterator->getEntity(pEntity, ZcDb::kForWrite);
		if (esGet != Zcad::eOk)
		{
			return;
		}

		if (!pEntity->isKindOf(ZcDbLine::desc()))
		{
			pEntity->close();
			return;
		}

		ZcDbLine* pLine = (ZcDbLine*)pEntity;
		pLine->setStartPoint(ZcGePoint3d(0, 100, 0));
		pLine->setEndPoint(ZcGePoint3d(100, 0, 0));

		pEntity->close();
	}
}

运行结果：

遍历模型空间中所有直线，如果直线为红色则修改其起点和终点坐标。

1. 创建圆

 void drawCircle()
{ 
	ZcGePoint3d centerPoint(0.0, 0.0, 0.0);
	double dRadius = 10.0;
	ZcGeVector3d x = zcdbHostApplicationServices()->workingDatabase()->ucsxdir();
	ZcGeVector3d y = zcdbHostApplicationServices()->workingDatabase()->ucsydir();
	ZcGeVector3d normal = x.crossProduct(y);
	normal.normalize();

	ZcDbCircle *pCircle = new ZcDbCircle(centerPoint, normal, dRadius);

	ZcDbBlockTable *pBlockTable;
	zcdbHostApplicationServices()->workingDatabase()->getSymbolTable(pBlockTable, ZcDb::kForRead);

	ZcDbBlockTableRecord *pBlockTableRecord;
	pBlockTable->getAt(ZCDB_MODEL_SPACE, pBlockTableRecord,	ZcDb::kForWrite);
	pBlockTable->close();

	ZcDbObjectId CircleId;
	pBlockTableRecord->appendZcDbEntity(CircleId, pCircle);

	pBlockTableRecord->close();
	pCircle->close();
}

运行结果：

在模型空间创建一个圆

1. 创建填充

void CreateHatchTest()
{
    ZcDbObjectIdArray ids, idsRet;
    if (RTNORM != SelectEntities(ids) || ids.length() == 0)
        return;

    std::map<ZcDbObjectId, double> idsMap;
    for (int i = 0; i < ids.length(); i++)
    {
        ZcDbEntity *pEnt = nullptr;
        double dArea = 0.0;
        Zcad::ErrorStatus es = zcdbOpenZcDbEntity(pEnt, ids[i], ZcDb::kForRead);
        ((ZcDbCurve*)pEnt)->getArea(dArea);

        idsMap.insert(std::pair<ZcDbObjectId, double>(ids[i], dArea));
        pEnt->close();
    }

    std::vector<std::pair<ZcDbObjectId, double>> vec;
    for (std::map<ZcDbObjectId, double>::iterator iter = idsMap.begin(); iter != idsMap.end(); iter++)
        vec.push_back(std::pair<ZcDbObjectId, double>(iter->first, iter->second));
    // Same type, boundary sorted by area
    std::sort(vec.begin(), vec.end(), [](std::pair<ZcDbObjectId, double>& a, std::pair<ZcDbObjectId, double>& b) {return a.second < b.second; });

    ZcDbHatch *pHatch = new ZcDbHatch();
    pHatch->setNormal(ZcGeVector3d::kZAxis);
    pHatch->setElevation(0.0);
    pHatch->setAssociative(ZSoft::kTrue);
    pHatch->setPattern(ZcDbHatch::kPreDefined, _T("SOLID"));
    zcedInitGet(NULL, _T("N O I"));
    TCHAR szRet[32] = { 0 };
    zcedGetKword(_T("Style type [Normal(N)/Outer(O)/Ignore(I)]:"), szRet);
    if (_tcscmp(_T("N"), szRet) == 0)
        pHatch->setHatchStyle(ZcDbHatch::kNormal);
    else if (_tcscmp(_T("O"), szRet) == 0)
        pHatch->setHatchStyle(ZcDbHatch::kOuter);
    else if (_tcscmp(_T("I"), szRet) == 0)
        pHatch->setHatchStyle(ZcDbHatch::kIgnore);

    int nDep = 0;
    std::vector<std::pair<ZcDbObjectId, double>>::iterator iter = vec.begin();
    for (; iter != vec.end(); iter++)
    {
        if (nDep == vec.size() - 1)  // The outermost boundary
        {
            ZcDbObjectIdArray arrExternalId;
            arrExternalId.append(iter->first);
            pHatch->appendLoop(ZcDbHatch::kExternal, arrExternalId);
        }
        else if (nDep == vec.size() - 2) // The boundary of the penultimate outer layer
        {
            ZcDbObjectIdArray arrOutermostId;
            arrOutermostId.append(iter->first);
            pHatch->appendLoop(ZcDbHatch::kOutermost, arrOutermostId);
        }
        else
        {
            ZcDbObjectIdArray arrDefaultId;
            arrDefaultId.append(iter->first);
            pHatch->appendLoop(ZcDbHatch::kDefault, arrDefaultId);
        }
        nDep++;
    }

    pHatch->evaluateHatch();
    ZcDbObjectId newId;
    AppendToModalSpace(curDoc()->database(), pHatch, newId);
    pHatch->close();
}
int SelectEntities(ZcDbObjectIdArray & entIdArray)
{
	zds_name ss;
	int ret = zcedSSGet(NULL, NULL, NULL, NULL, ss);
	if (ret == RTNORM)
	{
		ZSoft::Int32 len = 0;
		zcedSSLength(ss, &len);
		for (int i = 0; i < len; i++)
		{
			zds_name ent;
			ZcDbObjectId entId;
			zcedSSName(ss, i, ent);
			zcdbGetObjectId(entId, ent);
			if (entId.isValid())
			{
				entIdArray.append(entId);
			}
		}
		zcedSSFree(ss);
	}
	return ret;
}

Zcad::ErrorStatus AppendToModalSpace(ZcDbDatabase* db, ZcDbEntity* ent, ZcDbObjectId& oid)
{
	if (!db)
		return Zcad::eNoDatabase;
	if (!ent)
		return Zcad::eNullObjectPointer;

	Zcad::ErrorStatus stats = Zcad::eOk;
	ZcDbBlockTable* pBlockTable = NULL;
	stats = db->getBlockTable(pBlockTable, ZcDb::kForRead);
	if (pBlockTable)
	{
		ZcDbBlockTableRecord* pBTR = NULL;
		stats = pBlockTable->getAt(ZCDB_MODEL_SPACE, pBTR, ZcDb::kForWrite);
		pBlockTable->close();
		pBlockTable = NULL;
		if (pBTR)
		{
			stats = pBTR->appendZcDbEntity(oid, ent);
			pBTR->close();
		}
	}
	return stats;
}

		es = pTbl->mergeCells(0, 0, 0, 1);

		es = pTbl->setTextString(2, 1, sTxt);

		// Centers the table at (0,0)
		const ZcGePoint3d pt(-pTbl->width() / 2, pTbl->height() / 2, 0.0);
		es = pTbl->setPosition(pt);

		ZcDbObjectId Id;
		es = AppendEntity(pTbl, Id);
		if (es == Zcad::eOk)
			es = pTbl->close();
		else
		{
			delete pTbl;
			return;
		}


		// Further tests :
		ZcDbObjectPointer<ZcdbTable> pEnt(Id, ZcDb::kForRead);
		if ((es = pEnt.openStatus()) == Zcad::eOk)
		{
			zcutPrintf(_T("\nAfter appending the table to the drawing, pTbl->width () = %.1f and pTbl->height() = %.1f."), pTbl->width(), pTbl->height());
		}
}

Zcad::ErrorStatus AppendEntity(ZcDbEntity* pEntity, ZcDbObjectId &Id)
{
	Zcad::ErrorStatus es = Zcad::eNullPtr;
	if (!pEntity)
		return es;

	ZcDbDatabase* pDB = zcdbHostApplicationServices()->workingDatabase();
	if (!pDB)
		return es;

	ZcDbSymbolTable* pTbl = nullptr;
	if ((es = pDB->getBlockTable(pTbl, ZcDb::kForRead)) != Zcad::eOk)
		return es;

	ZcDbSymbolTableRecord* pRec = nullptr;
	if ((es = pTbl->getAt(ZCDB_MODEL_SPACE, pRec, ZcDb::kForWrite)) == Zcad::eOk)
	{
		es = ZcDbBlockTableRecord::cast(pRec)->appendZcDbEntity(Id, pEntity);
		pRec->close();
	}
	pTbl->close();

	return es;
}
