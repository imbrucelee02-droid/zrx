双击扩展协议

双击扩展协议可以让用户在双击实体时，执行自定义的操作，例如弹出属性对话框等，常与自定义实体结合使用。本节将简单介绍如何给圆添加自定义的双击协议。

创建自定义双击类

在工程中创建新类，设置基类为ZcDbDoubleClickEdit，重写startEdit和finishEdit两个虚函数：

// CDoubleClick.h
class CDoubleClick :
    public ZcDbDoubleClickEdit
{
public:
    virtual void startEdit(ZcDbEntity *pEnt, ZcGePoint3d pt);
    virtual void finishEdit(void);
};

// CDoubleClick.cpp
void CDoubleClick::startEdit(ZcDbEntity * pEnt, ZcGePoint3d pt)
{
    ZcApDocument *pDoc = zcDocManager->curDocument();
    zcDocManager->lockDocument(pDoc);
    // 判断双击实体是否为圆
    if (pEnt->isKindOf(ZcDbCircle::desc()))
    {
        ZcDbCircle *c = ZcDbCircle::cast(pEnt);
        if (c->upgradeOpen() != Zcad::eOk)
        {
            zcutPrintf(_T("\ncan't open entity"));
            zcDocManager->unlockDocument(pDoc);
            return;
        }
        zds_alert(_T("this is a circle"));
        c->close();
    }
    zcDocManager->unlockDocument(pDoc);
    return;
}

void CDoubleClick::finishEdit(void)
{
}

注册双击事件

CDoubleClick *pEntity;

void initapp()
{
    // 注册双击类
    ZcDbDoubleClickEdit::rxInit();
    zcrxBuildClassHierarchy();

    pEntity = new CDoubleClick();
    // 对指定的实体类添加双击协议
    ZcDbCircle::desc()->addX(ZcDbDoubleClickEdit::desc(), pEntity);

}

void unloadapp()
{
    zcedRegCmds->removeGroup(cmd_group_name);

    ZcDbEntity::desc()->delX(ZcDbDoubleClickEdit::desc());
    delete pEntity;
}
