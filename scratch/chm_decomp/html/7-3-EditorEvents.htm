编辑器反应器

编辑器反应器中常用反应器有打开或关闭图纸、执行命令过程等。

常用编辑器反应器

事件 功能概述
beginDwgOpen 即将打开的文档
commandWillStart 即将执行的命令
commandEnded 命令执行完毕
commandCancelled 命令被取消

定义反应器内容

在工程中新建类，设置基类为ZcEditorReactor，在类中添加所需的事件虚函数。 

//EdReactor.h
class EdReactor : public ZcEditorReactor
{
public:
    ZCRX_DECLARE_MEMBERS(EdReactor);

    EdReactor();
    virtual ~EdReactor();

    virtual void commandWillStart(const ZTCHAR* cmdStr);
    virtual void commandEnded(const ZTCHAR* cmdStr);
    virtual void commandCancelled(const ZTCHAR* cmdStr);
    virtual void beginDwgOpen(const ZTCHAR* filename);
};

// EdReactor.cpp
ZCRX_NO_CONS_DEFINE_MEMBERS(EdReactor, ZcEditorReactor);

EdReactor::EdReactor()
{
}

EdReactor::~EdReactor()
{
}

void EdReactor::commandWillStart(const ZTCHAR* cmdStr)
{
    zcutPrintf(_T("\n%s will start!"), cmdStr);
}

void EdReactor::commandEnded(const ZTCHAR* cmdStr)
{
    zcutPrintf(_T("\n%s ended!"), cmdStr);
}

void EdReactor::commandCancelled(const ZTCHAR* cmdStr)
{
    zcutPrintf(_T("\n%s cancelled!"), cmdStr);
}

void EdReactor::beginDwgOpen(const ZTCHAR* filename)
{
    zcutPrintf(_T("\n%s beginDwgOpen!"), filename);
}

添加和删除编辑器反应器

static EdReactor* s_g_editorreactor = nullptr;

void initApp()
{ 
	EdReactor::rxInit();
	zcrxBuildClassHierarchy();
	zcedRegCmds->addCommand(_T("EDITORREACTOR_COMMANDS"), _T("AddEditorReactor"), _T("AddEditorReactor"), ZCRX_CMD_MODAL, AddEditorReactor);
	zcedRegCmds->addCommand(_T("EDITORREACTOR_COMMANDS"), _T("RemoveEditorReactor"), _T("RemoveEditorReactor"), ZCRX_CMD_MODAL, RemoveEditorReactor);
}

void unloadApp()
{ 
	RemoveEditorReactor();

	deleteZcRxClass(EdReactor::desc());
	zcedRegCmds->removeGroup(_T("EDITORREACTOR_COMMANDS"));
}

void AddEditorReactor()
{
	if (!s_g_editorreactor)
	{
		s_g_editorreactor = new EdReactor();
		zcedEditor->addReactor(s_g_editorreactor);
		zcutPrintf(_T("\nEdReactor is on..."));
	}
}

void RemoveEditorReactor()
{
	if(s_g_editorreactor)
	{
		zcedEditor->removeReactor(s_g_editorreactor);
		delete s_g_editorreactor;
		s_g_editorreactor = NULL;
		zcutPrintf(_T("\nEdReactor is off..."));
	}
}
